import pandas as pd
from datetime import datetime
import openpyxl
from openpyxl.utils.dataframe import dataframe_to_rows

# ==========================================
# 1. PENGATURAN AWAL (Ubah bagian ini jika diperlukan)
# ==========================================
# Nama file CSV yang Anda dapatkan dari mesin absen
NAMA_FILE_CSV = 'ekspor_csv Mei 26.csv'

# Nama file Excel yang ingin dihasilkan
NAMA_FILE_EXCEL = 'Laporan_Absensi_Brahmanda_Mei_2026.xlsx'

# Data Pribadi untuk di Kop Surat
NAMA_PEGAWAI = 'I Made Brahmanda Setyadi, S.Kom'
OPD_PEGAWAI = 'Dinas Komunikasi dan Informatika'
PROJEK_PEGAWAI = 'Open SID (Sistem Informasi Desa) Kab. Badung'
ROLE_PEGAWAI = 'Full Stack Web Developer'

# ==========================================
# 2. PROSES MEMBACA DAN MERAPIKAN DATA
# ==========================================
print("Sedang membaca data CSV...")
# Membaca data menggunakan pemisah titik koma (;)
df = pd.read_csv(NAMA_FILE_CSV, sep=';')

# Mencari baris data khusus untuk nama Anda
data_saya = df[df['nama'] == NAMA_PEGAWAI]

# Kita siapkan daftar kosong untuk menampung data yang sudah rapi
hasil_rekap = []

if data_saya.empty:
    print(f"Maaf, nama '{NAMA_PEGAWAI}' tidak ditemukan di file CSV. Mohon cek huruf besar/kecilnya.")
else:
    print("Nama ditemukan! Sedang memproses tanggal...")
    # Mengambil semua nama kolom mulai dari kolom ke-3 (karena kolom 1 nama, kolom 2 nik)
    kolom_tanggal = df.columns[2:]

    for tanggal_str in kolom_tanggal:
        # Mengambil isi absen di tanggal tersebut
        absen = data_saya.iloc[0][tanggal_str]

        # Ubah teks tanggal (DD-MM-YYYY) menjadi objek penanggalan komputer
        tanggal_obj = datetime.strptime(tanggal_str, '%d-%m-%Y')
        nama_hari = tanggal_obj.strftime('%A')
        # Ubah format menjadi YYYY-MM-DD seperti di template excel
        tanggal_format_baru = tanggal_obj.strftime('%Y-%m-%d')

        # Nilai bawaan jika kosong
        jam_masuk = '-'
        jam_pulang = '-'
        durasi_kerja = '-'
        keterangan = '-'

        # Tentukan apakah hari tersebut akhir pekan
        if nama_hari == 'Saturday':
            keterangan = 'Sabtu'
        elif nama_hari == 'Sunday':
            keterangan = 'Minggu'
        else:
            # Jika bukan akhir pekan, cek apakah ada data jam masuk/pulang
            # pd.notna() mengecek apakah data tersebut tidak kosong (NaN)
            if pd.notna(absen) and str(absen).strip() != '':
                try:
                    # Pisahkan teks "Masuk - Pulang" (contoh: "08:05:43 - 16:05:13")
                    jam_masuk, jam_pulang = str(absen).split(' - ')

                    # Menghitung durasi jam kerja
                    waktu_masuk = datetime.strptime(jam_masuk, '%H:%M:%S')
                    waktu_pulang = datetime.strptime(jam_pulang, '%H:%M:%S')
                    selisih = waktu_pulang - waktu_masuk

                    # Hitung jam dan menit dari total selisih detik
                    jam = selisih.seconds // 3600
                    menit = (selisih.seconds // 60) % 60
                    durasi_kerja = f'{jam} jam {menit} menit'
                    keterangan = 'Hadir'
                except Exception as e:
                    # Jika formatnya aneh atau tidak lengkap
                    keterangan = 'Format Error/Tidak Lengkap'
            else:
                # Jika hari kerja tapi datanya kosong
                keterangan = 'Tidak Hadir / Cuti'

        # Masukkan hasil per hari ke dalam daftar kita
        hasil_rekap.append({
            'Tanggal(s)': tanggal_format_baru,
            'Jam Masuk': jam_masuk,
            'Jam Pulang': jam_pulang,
            'Durasi Kerja': durasi_kerja,
            'Keterangan': keterangan
        })

    # ==========================================
    # 3. MEMBUAT FILE EXCEL SESUAI TEMPLATE
    # ==========================================
    print("Menulis ke dalam file Excel...")
    df_hasil = pd.DataFrame(hasil_rekap)

    # Bikin dokumen Excel baru
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "Sheet1" # Nama sheet di bawah

    # Tulis teks Kop Surat dan Biodata persis seperti template Anda
    kop_surat = """PEMERINTAH KABUPATEN BADUNG
DINAS KOMUNIKASI DAN INFORMATIKA
PUSAT PEMERINTAHAN MANGUPRAJA MANDALA
Jln Raya Sempidi, Mengwi – Kabupaten Badung (80351)
Telp. (0361) 419888 Fax (0361) 419888
Website : www.diskominfo.badungkab.go.id"""

    ws.cell(row=1, column=3, value=kop_surat)
    ws.cell(row=8, column=2, value="Absensi Kehadiran")

    # Biodata baris pertama
    ws.cell(row=10, column=2, value="OPD :")
    ws.cell(row=10, column=3, value=OPD_PEGAWAI)
    ws.cell(row=10, column=5, value="Nama :")
    ws.cell(row=10, column=6, value=NAMA_PEGAWAI)

    # Biodata baris kedua
    ws.cell(row=11, column=2, value="Projek :")
    ws.cell(row=11, column=3, value=PROJEK_PEGAWAI)
    ws.cell(row=11, column=5, value="Role :")
    ws.cell(row=11, column=6, value=ROLE_PEGAWAI)

    # Menulis Header Tabel data (Tanggal, Jam Masuk, dll)
    header_tabel = ['Tanggal(s)', 'Jam Masuk', 'Jam Pulang', 'Durasi Kerja', 'Keterangan']
    for col_num, nama_kolom in enumerate(header_tabel, 2): # Mulai di kolom B (indeks 2)
        ws.cell(row=13, column=col_num, value=nama_kolom)

    # Menulis baris-baris data dari hasil hitungan kita di atas
    baris_sekarang = 14
    for r in dataframe_to_rows(df_hasil, index=False, header=False):
        for col_num, nilai in enumerate(r, 2):
             ws.cell(row=baris_sekarang, column=col_num, value=nilai)
        baris_sekarang += 1

    # Simpan file Excel-nya
    wb.save(NAMA_FILE_EXCEL)
    print(f"Hore Selesai! Silakan cek file '{NAMA_FILE_EXCEL}' di folder Anda.")
